\section{مقدمه و انگیزه}
%==================================================

\begin{frame}{سیستم نوع محض}

\textbf{تعریف:} یک سیستم نوع محض با سه‌تایی $(\mathcal{S},\mathcal{A},\mathcal{R})$ مشخص می‌شود:

\begin{itemize}
  \item $\mathcal{S}$: مجموعه رده‌ها
  \item $\mathcal{A} \subseteq \mathcal{S} \times \mathcal{S}$: مجموعه اصول
  \item $\mathcal{R} \subseteq \mathcal{S} \times \mathcal{S} \times \mathcal{S}$: مجموعه قوانین
\end{itemize}

\vfil
\begin{align*}
&T ::= \mathcal{S} \mid V \mid \lambda V^T.T \mid \Pi V^T.T \mid TT\\ \\
&(\lambda x^A.M)N \rightarrow_{\beta} M[N/x]
\end{align*}

\end{frame}

\begin{frame}{مسئله اصلی}

{\small
  \begin{itemize}
      \item \textbf{فرم نرمال ($\mathrm{NF}$):} عبارتی که قابل کاهش نباشد.
      \begin{align*}
        \lambda x^A.x
      \end{align*}
      \item \textbf{نرمال‌سازی ضعیف ($\mathrm{WN}$):} وجود مسیر کاهش به فرم نرمال برای هر عبارت نوع‌پذیر.
      \begin{align*}
        M \rightarrow_{\beta} \dots \rightarrow_{\beta} N
      \end{align*}
      \item \textbf{نرمال‌سازی قوی ($\mathrm{SN}$):} عدم وجود دنباله کاهش نامتناهی برای هر عبارت نوع‌پذیر.
      \begin{align*}
        M \rightarrow_{\beta} N \rightarrow_{\beta} \dots
      \end{align*}
  \end{itemize}
}

\textbf{مسئله نرمال‌سازی (حدس \lr{BGK})}\\
در هر سیستم نوع محض؛
نرمال‌سازی ضعیف
\pmb{$\overset{?}{\impliedby}$}
نرمال‌سازی قوی
\hfil\\

{\scriptsize این حدس بیش از سه دهه است که برای سیستم‌های وابسته باز مانده است.}

\end{frame}

\begin{frame}{چرا این مسئله دشوار است؟}
 
در سیستم‌های نوع محض وابسته،
\textbf{نوع}
یک عبارت، می‌تواند به
\textbf{عبارت}
دیگری وابسته باشد.

\begin{itemize}
  {\small
  \item کاهش روی نوع اثر می‌گذارد.
  \item رفتار محاسباتی و ساختار منطقی در هم تنیده می‌شود.
  \item روش‌های کلاسیک اثبات نرمال‌سازی (کاهش‌پذیری، استدلال‌های ترتیبی) به‌سادگی قابل اعمال نیستند.
  }
\end{itemize}

\hfil{}\\
\textbf{مثال:} در استنتاج زیر، $P\,Q$ یک نوع است که به عبارت $Q$ وابسته است.
\begin{prooftree}  
\AxiomC{$\Gamma \vdash Q \;:\; A$}
\AxiomC{$\Gamma \vdash P \;:\; \Pi x^A.s$}
\BinaryInfC{$\Gamma \vdash P\,Q \;:\; s$}
\end{prooftree}
 
\end{frame}

\begin{frame}{تلاش‌های پیشین}

ایده کلی: برای اثبات نرمال‌سازی قوی یک سیستم وابسته، آن را به سیستمی ساده‌تر که نرمال‌سازی قوی آن شناخته‌شده است، \textbf{ترجمه} کنیم.

{\small
\begin{itemize}
\item \lr{Harper} و همکاران (۱۹۹۳): ترجمه‌ای از $\lambda P$ به $\lambda_\to$
\item \lr{Geuvers} و \lr{Nederhof} (۱۹۹۱): تعمیم به ترجمه‌ای از $\lambda C$ به $\lambda \omega$
\item \lr{Barthe} و همکاران (۲۰۰۱): تعریف سیستم‌های نوع محض غیروابسته تعمیم‌یافته
\end{itemize}
}

\vfil
تمام این ترجمه‌ها، حذف قانون وابسته را در سیستم مربوطه انجام می‌دهند.\\
\vfil
\textbf{اما}: این روش‌ها به‌طور کامل به همه سیستم‌های نوع محض تعمیم داده نشده‌اند.

\end{frame}

\begin{frame}{سیستم‌های نوع محض لایه‌ای}

\textbf{ایده اصلی \lr{Mull}:}

{\small
  \begin{itemize}
    \item مرتب کردن $\mathcal{S}$ و محدود کردن $\mathcal{A}$
    \begin{equation*}
    s_1 \overset{\mathcal{A}}{\longrightarrow} s_2 \overset{\mathcal{A}}{\longrightarrow} s_3 \overset{\mathcal{A}}{\longrightarrow} \dots \overset{\mathcal{A}}{\longrightarrow} s_n
    \end{equation*}
    \item انتساب یک درجه‌ به هر عبارت
    \begin{equation*}
    \operatorname{deg}: \mathrm{T} \to \mathbb{N}
    \end{equation*}
    \item استفاده از این ساختار خوش‌تعریف، به منظور تعریف یک \textbf{ترجمه حذف وابستگی}
  \end{itemize}
}

\end{frame}

\begin{frame}{تعریف سیستم نوع محض $n$-لایه‌ای}

یک سیستم نوع محض، $n$-لایه‌ای است هرگاه:

\begin{align*}
\mathcal{S} &= \{ s_i \mid i \in \{1,\dots,n\} \}\\
\mathcal{A} &= \{ (s_i,s_{i+1}) \mid i \in \{1,\dots,n{-}1\} \}\\
\mathcal{R} &\subset \{ (s,s',s') \mid (s,s') \in \mathcal{S} \times \mathcal{S} \}
\end{align*}

\vfil
{\small
\begin{itemize}
  \item $n=0$: سیستم نوع محض تهی
  \item $n=2$: معادل کل مکعب لامبدا (هر هشت راس)
\end{itemize}
}

\end{frame}

\begin{frame}{وابستگی در سیستم نوع محض لایه‌ای}

\textbf{قاعده تشکیل نوع تابع:}
\begin{prooftree}
\AxiomC{$\Gamma,x:A \vdash B:s_j$}
\AxiomC{$\Gamma \vdash A:s_i$}
\LeftLabel{$\quad (s_i,s_j,s_j) \in \mathcal{R}$}
\BinaryInfC{$\Gamma \vdash (\Pi x^A.B):s_j$}
\end{prooftree}

در استنتاج فوق اگر
$i < j$،
آنگاه انواعی به وجود می‌آیند که به عباراتی از درجه پایین‌تر وابسته‌اند.

\end{frame}

\begin{frame}{نسخه غیروابسته سیستم نوع محض لایه‌ای}

\textbf{نسخه غیروابسته} یک سیستم نوع محض لایه‌ای
$\lambda S$،
که با
$\lambda S^*$
نشان داده می‌شود، سیستمی با مشخصات زیر است:
\begin{align*}
S_{\lambda S^*} &= S_{\lambda S}\\
\mathcal{A}_{\lambda S^*} &= \mathcal{A}_{\lambda S}\\
\mathcal{R}_{\lambda S^*} &= \{(s_i,s_j,s_j) \in \mathcal{R}_{\lambda S} \mid (i \ge j)\}
\end{align*}

\end{frame}

\begin{frame}{ترجمه حذف وابستگی \lr{Mull}}

{\Large
\begin{equation*}
\lambda S \xrightarrow{\hspace{0.5cm}\rho_i \;,\; \gamma\hspace{0.5cm}} \lambda S^*
\end{equation*}
}

\begin{itemize}
  \item \textbf{تابع ترجمه سطح نوع $\bm{\rho_i}$:} انواع را به رده‌ای پایین‌تر نگاشت می‌کند.
  \item \textbf{تابع ترجمه سطح عبارت $\bm{\gamma}$:} عبارات را به یک عبارت غیروابسته تبدیل می‌کند.
\end{itemize}


\end{frame}

\begin{frame}{سه لم کلیدی ترجمه}

{\small
\begin{enumerate}
  \item \textbf{با جانشینی جابه‌جا می‌شوند:}
  \begin{align*}
  \rho_i(A[B/x]) &= \rho_i(A)[\rho_i(B)/x]\\
  \gamma(A[B/{}^{s_{j}}x]) &= \gamma(A) [\rho_{j-2}(B)/{}^{s_{j}}x] \dots [\rho_{0}(B)/{}^{s_{2}}x] [\gamma(B)/{}^{s_{1}}x]
  \end{align*}
  \item \textbf{مسیر کاهش را حفظ می‌کنند:}
  \begin{align*}
  A \twoheadrightarrow_\beta B &\implies \rho_i(A) \twoheadrightarrow_\beta \rho_i(B)\\
  A \to_\beta B &\implies \gamma(A) \twoheadrightarrow^+_\beta \gamma(B)
  \end{align*}
  \item \textbf{نوع را خوش‌تعریف نگاشت می‌کنند:}
  \begin{align*}
  \Gamma \vdash_{\lambda S} A:B &\implies \rho_i(\Gamma) \; \vdash_{\lambda S^*} \; \rho_i(A):\rho_i(B)\\
  \Gamma \vdash_{\lambda S} A:B &\implies \Delta,\; \rho_0(\Gamma) \; \vdash_{\lambda S^*} \; \gamma(A) : \rho_0(B)
  \end{align*}
\end{enumerate}
}

\end{frame}

\begin{frame}{نتیجه اصلی \lr{Mull}}

\textbf{قضیه:} فرض کنید $\lambda S$ یک سیستم نوع محض لایه‌ای است که \textbf{کامل} است، آنگاه:
\begin{align*}
\lambda S \in \mathrm{SN} \iff \lambda S^* \in \mathrm{SN}
\end{align*}

\vfil
\textbf{نتیجه:}\\
حدس \lr{BGK} برای خانواده‌ مشخصی از سیستم‌های نوع محض لایه‌ای برقرار است و این اثبات را می‌توان در \textbf{حساب پئانو} انجام داد.

\vfil
{\scriptsize این نتیجه، یکی از بزرگ‌ترین پیشرفت‌های اخیر در مسیر اثبات حدس \lr{BGK} برای سیستم‌های وابسته است.}

\end{frame}

  \end{align*}
  }

    | Γ0 M0 N0 A0 B0 s1a HM IHM HN IHN

\begin{frame}[fragile]{نمونه‌: صوری‌سازی ترجمه سطح نوع}
\begin{latin}
\setmonofont{DejaVu Sans Mono}
\begin{minted}[fontsize=\tiny]{coq}
Fixpoint rho (i : nat) (M : term) : term :=
  match M with
  | t_sort s =>
      match i with
      | 0 => t_fvar (sort_of 2) zero_var
      | _ => t_sort (sort_of i)
      end
  | t_bvar s0 k => t_bvar s0 k
  | t_fvar s0 x => t_fvar (sort_of (i + 2)) (retag x (sort_of (i + 2)))
  | t_pi s A B =>
      match le_lt_dec (i + 1) (deg A) with
      | left _  => rho_pi_tel rho A B i (deg A - 1 - i)
      | right _ => rho i B
      end
  | t_lam s A M' =>
      match le_lt_dec (i + 2) (deg A) with
      | left _  => rho_lam_tel rho A M' i (deg A - 1 - (i + 1))
      | right _ => rho i M'
      end
  | t_app M' N =>
      match le_lt_dec (i + 1) (deg N) with
      | left _  => rho_app_tel rho M' N i (deg N - 1 - i)
      | right _ => rho i M'
      end
  end.
\end{minted}
\end{latin}
\end{frame}

\begin{frame}[fragile]{نمونه‌: صوری‌سازی لم جابه‌جایی ترجمه سطح نوع با جانشینی}
\begin{latin}
\setmonofont{DejaVu Sans Mono}
\begin{minted}[fontsize=\scriptsize]{coq}
Lemma rho_commutes_substitution :
  forall i, 0 <= i <= n ->
  forall x s, x <> zero_var -> 1 <= index_of s <= n ->
  forall M, deg M >= i + 1 -> only_tagged x s M ->
  forall N, deg N = index_of s - 1 -> lc N ->
  rho i (M ⁅ x ≔ N ⁆) = rho_subst M N x s i.
Proof.
  intros i.
  induction i as [i IHouter] using
    (well_founded_induction (Wf_nat.well_founded_ltof nat (fun i => n - i))).
  intros Hi x s Hxnz Hx M.
  induction M as [s0 | s_m m | sy y | D IHD C IHC | s1 C IHC D IHD | s1 C IHC D IHD]; intros Hdeg Honly N HdegN HlcN.

  (* continued! *)
\end{minted}
\end{latin}
\end{frame}

\begin{frame}[fragile]{نمونه‌: صوری‌سازی قضیه اصلی}
\begin{latin}
\setmonofont{DejaVu Sans Mono}
\begin{minted}[fontsize=\scriptsize]{coq}
Theorem lambdaS_SN_iff_lambdaS_star_SN :
  system_strongly_normalizing derivable <-> 
  system_strongly_normalizing derivable_star.
Proof.
  split.
  - (* λS SN -> λS∗ SN *)
    intros HSN M HMstar.
    apply HSN.
    apply derivable_star_implies_derivable.
    exact HMstar.
  - (* λS∗ SN -> λS SN *)
    intros HSNstar M HM.
    apply sn_reflection.
    apply HSNstar.
    apply gamma_commutes_typing_derivable.
    exact HM.
Qed.
\end{minted}
\end{latin}
\end{frame}

\begin{frame}{آنچه در این پایان‌نامه صوری‌سازی شد}

{\small
  \begin{itemize}
  \item \textbf{تعریف کامل سیستم‌های نوع محض}، مانند:
    \begin{itemize}
    \item نحو عبارات، جانشینی و کاهش $\beta$
    \item نرمال‌سازی ضعیف و قوی، استنتاج نوع
    \item خواص و لم‌های سیستم
    \end{itemize}
  \item \textbf{تعریف کامل سیستم‌های نوع محض لایه‌ای}، مانند:
    \begin{itemize}
    \item تابع درجه و لم طبقه‌بندی
    \item توابع ترجمه سطح نوع ($\rho_i$) و سطح عبارت ($\gamma$)
    \item لم‌های حفظ جانشینی، حفظ مسیر کاهش و حفظ نوع
    \end{itemize}
  \item \textbf{قضیه اصلی}: $\lambda S \in \mathrm{SN} \iff \lambda S^* \in \mathrm{SN}$
  \end{itemize}
}

\end{frame}

\begin{frame}[fragile]{محدودیت‌ها}
به دلیل
\textbf{محدودیت زمانی}،
برخی گام‌های اثبات لم‌های مربوط به ترجمه حذف وابستگی، به‌صورت \lr{Axiom} در \lr{Coq} فرض شده‌اند.

\begin{latin}
\setmonofont{DejaVu Sans Mono}
\begin{minted}[fontsize=\scriptsize]{coq}
Axiom gamma_app_tel_commutes_subst : forall MG C B x s KC,
  x <> zero_var -> 1 <= index_of s <= n ->
  deg B = index_of s - 1 -> lc B -> only_tagged x s C ->
  (gamma_subst_tel (gamma_app_tel MG C KC) B x (index_of s - 1))
    ⁅ retag x (sort_of 1) ≔ gamma B ⁆
  = gamma_app_tel
      ((gamma_subst_tel MG B x (index_of s - 1))
         ⁅ retag x (sort_of 1) ≔ gamma B ⁆)
      (C ⁅ x ≔ B ⁆) KC.
\end{minted}
\end{latin}

\end{frame}

\begin{frame}{دستاوردهای پایان‌نامه}

\begin{itemize}
  \item \textbf{اولین} صوری‌سازی کاملاً ماشینی قضیه اصلی \lr{Mull}
  \item اثبات ماشینی تمام لم‌های واسطه (جانشینی، حفظ مسیر کاهش، حفظ نوع)
  \item توسعه یک \textbf{کتابخانه باز و قابل گسترش} در \lr{Coq}
  \item بستری برای تعمیم به سیستم‌های وابسته‌تر مانند \textbf{حساب ساختمان‌ها}
  \item گامی مهم در مسیر اثبات حدس \lr{BGK} برای سیستم‌های وابسته
\end{itemize}

\end{frame}

  \item صوری‌سازی امکان اجرای اثبات در حساب پئانو
