\section{مقدمه‌}

در این فصل مفاهیم و تعاریف پایه‌ای که در ادامهٔ پایان‌نامه مورد نیاز هستند معرفی می‌شوند.
هدف اصلی فراهم ساختن یک زمینه مفهومی و نمادشناسی واحد برای خواننده است
تا مطالعهٔ فصل‌های بعدی که به توصیف سیستم‌های نوع محض لایه‌ای و صوری‌سازی آن‌ها در
\lr{Coq}
می‌پردازند، ممکن گردد.
در این فصل تمرکز بر تعریف نحوها، روابط کاهش، قضایا و لم‌های کلاسیکِ مورد استفاده در این پایان‌نامه است؛
اثبات‌های تفصیلی اغلب به منابع استاندارد مانند
\cite{barendregt} و \cite{geuvers1993}
ارجاع داده خواهد شد.

\section{حساب لامبدا بدون نوع}

\subsection{نحو}

حساب لامبدا به عنوان یک دستگاه انتزاعی برای نمایش محاسبه و تبدیلِ عباراتِ تابعی تعریف می‌شود.

\begin{defn}    
مجموعه عبارات
$\Lambda$
با قاعده بازگشتی زیر تعریف می‌گردد:

\begin{equation}
M::=V|\lambda V.M|MM
\end{equation}

که در آن
$V=\{x,y,z,...\}$
مجموعه متغیرها،
$\lambda V.M$
انتزاع و
$MM$
کاربرد است.
\end{defn}

\subsection{متغیرهای آزاد و بسته}

\begin{defn}
برای یک عبارت
$M$،
مجموعه متغیرهای آزاد آن که با
$FV(M)$
نشان داده می‌شود، به صورت بازگشتی به شکل زیر تعریف می‌گردد
\begin{align}
FV(x)&=\{x\}\\
FV(\lambda x.M)&=FV(M)-\{x\}\\
FV(MN)&=FV(M) \cup FV(N)
\end{align}
\end{defn}

\begin{defn}
گوییم متغیر
$x$
در
$M$
آزاد است، هرگاه
$x \in FV(M)$.
گوییم متغیر
$x$
در
$M$
بسته است، هرگاه
$x \notin FV(M)$.
\end{defn}

\subsection{جانشینی}

جانشینی با نماد
$M[N/x]$
نشان داده می‌شود و معنای آن جایگزینی همه رخدادهای آزاد
$x$
در
$M$
با عبارت
$N$
است.

\begin{defn}
جانشینی به صورت بازگشتی زیر تعریف می‌شود
\begin{align}
x[N/x] &= N\\
y[N/x] &= y & (x \neq y) \\
PQ[N/x]&=(P[N/x])(Q[N/x])\\
(\lambda x.M)[N/x] &= \lambda x.M\\
(\lambda y.M)[N/x] &= \lambda y.(M[N/x]) & (x \neq y)
\end{align}
\end{defn}

\subsection{تبدیل $\alpha$}

تبدیل
$\alpha$
به ما اجازه می‌دهد نام متغیرهای بسته را بدون تغییر معنای عبارت تغییر دهیم.

\begin{defn}
اگر
$M$
یک عبارت باشد و
$x \notin FV(M)$،
آنگاه تبدیل
$\alpha$
به صورت زیر بیان می‌شود
\begin{equation}
\lambda x.M \equiv_{\alpha} \lambda y.M[y/x]
\end{equation}
\end{defn}

\subsection{کاهش $\beta$}

\begin{defn}
کاهش
$\beta$
قاعده اصلی محاسبه در حساب لامبدا است و به صورت زیر تعریف می‌شود
\begin{equation}
(\lambda x.M)N \rightarrow_{\beta} M[N/x]
\end{equation}
که بستار بازتابی تعدی آن با نماد
$\twoheadrightarrow_{\beta}$
نشان داده می‌شود.
\end{defn}

\subsection{مسیرهای کاهش و فرم نرمال}

\begin{defn}
عبارت
$M$
در فرم نرمال است، اگر هیچ گامی از کاهش
$\beta$
بر آن قابل اعمال نباشد.
یعنی
$\nexists N. M \rightarrow_{\beta} N$.
\end{defn}

\begin{defn}
یک مسیر کاهش، دنباله‌ای به شکل
$M_0 \rightarrow_{\beta} M_1 \rightarrow_{\beta} ... \rightarrow_{\beta} M_n$
است که می‌تواند متناهی یا نامتناهی باشد.
\end{defn}

\subsection{نرمال‌سازی ضعیف و قوی}

\begin{defn}
عبارت
$M$
دارای خاصیت نرمال‌سازی ضعیف است، اگر حداقل یک مسیر کاهش به یک فرم نرمال داشته باشد.
یعنی
$\exists N.\;(M \twoheadrightarrow_{\beta} N \land N \in \mathrm{NF})$
\end{defn}

\begin{defn}
عبارت
$M$
دارای خاصیت نرمال‌سازی قوی است، اگر هیچ مسیر کاهش نامتناهی‌ای از آن آغاز نشود.
\end{defn}

\section{سیستم‌های کاهش انتزاعی}

سیستم‌های کاهش انتزاعی\footnote{\lr{Abstract Reduction Systems} یا به اختصار \lr{ARS}}
چارچوبی ریاضی برای مطالعه روابط کاهش در سیستم‌های محاسباتی هستند.
بسیاری از مفاهیم نظریه انواع
(از جمله نرمال‌سازی ضعیف و قوی، همگرایی و یکتایی فرم نرمال)
به‌طور طبیعی در این چارچوب بیان می‌شوند.
در این بخش تعریف رسمی و ویژگی‌های بنیادی این چارچوب را بیان می‌کنیم.

\subsection{تعریف سیستم کاهش انتزاعی}

\begin{defn}
یک سیستم کاهش انتزاعی، یک زوج
$(A,\rightarrow)$
است که در آن:
\begin{itemize}
\item 
$A$
مجموعه‌ای از اشیا است.
\item
$\rightarrow$
یک رابطه دودویی روی
$A$
است که رابطه کاهش نام دارد.
\end{itemize}
\end{defn}

برای مثال در حساب لامبدا، مجموع عبارات
$\Lambda$
همراه با رابطه کاهش
$\beta$
یک
ARS
را تشکیل می‌دهند.

\subsection{دنباله و بستار کاهش}

\begin{defn}
دنباله کاهش، زنجیره‌ای از کاهش‌ها به شکل
$a_0 \rightarrow a_1 \rightarrow ... \rightarrow a_n$
است، که می‌تواند متناهی یا نامتناهی باشد.
\end{defn}

\begin{defn}
بستار بازتابی-تعدی رابطه
$\rightarrow$
را با
$b \rightarrow^* a$
نشان می‌دهیم که یعنی
$b = a$
یا
دنباله کاهشی از
$b$
به
$a$
وجود دارد.
\end{defn}

\subsection{فرم نرمال و نرمال‌سازی}

\begin{defn}
شی
$a$
یک فرم نرمال است اگر
$\nexists b.\;a \rightarrow b$.
\end{defn}

\begin{defn}
شی
$a$
دارای خاصیت نرمال‌سازی ضعیف است، اگر
$\exists b.\;(a \rightarrow^* b \land b \in \mathrm{NF})$.
\end{defn}

\begin{defn}
شی
$a$
دارای خاصیت نرمال‌سازی قوی است، اگر هیچ دنباله کاهش نامتناهی از آن آغاز نشود.
\end{defn}

\begin{rem}
بدیهی است که
\lr{$\text{Strong Normalization} \implies \text{Weak Normalization}$}،
اما عکس آن همیشه برقرار نیست.
\end{rem}

\subsection{همگرایی}

\begin{defn}[همگرایی]
سیستم کاهش انتزاعی
$(A,\rightarrow)$
همگراست هرگاه
\begin{equation*}
\forall a,b_1,b_2.\;(a \rightarrow^* b_1 \land a \rightarrow^* b_2) \implies (\exists c.\; (b_1 \rightarrow^* c \land b_2 \rightarrow^* c))
\end{equation*}
به بیان ساده؛ هر دو مسیر کاهش که از یک شی منشعب می‌شوند، دوباره به شی مشترکی می‌رسند.
\end{defn}

نتایجی مانند یکتایی فرم نرمال از قضیه پیش رو ثابت می‌شوند.

\begin{thm}[قضیه چرچ-راسر]
سیستم کاهش انتزاعی
$(A,\rightarrow)$
همگراست اگر و تنها اگر دارای خاصیت چرچ-راسر باشد؛ یعنی
\begin{equation*}
\forall b_1,b_2.\;(b_1 = b_2) \implies (\exists c.\; (b_1 \rightarrow^* c \land b_2 \rightarrow^* c))
\end{equation*}
\end{thm}

اثبات همگرایی معمولاً دشوار است، اما یک ویژگی محلی ساده‌تر وجود دارد.

\begin{defn}[همگرایی محلی]
سیستم کاهش انتزاعی
$(A,\rightarrow)$
همگرای محلی است اگر
\begin{equation*}
\forall a,b_1,b_2.\;(a \rightarrow b_1 \land a \rightarrow b_2) \implies (\exists c.\; (b_1 \rightarrow^* c \land b_2 \rightarrow^* c))
\end{equation*}
\end{defn}

\begin{lmm}[لم نیومن]
اگر یک سیستم کاهش انتزاعی،همگرای محلی و دارای خاصیت نرمال‌سازی قوی باشد، آنگاه همگراست.
\end{lmm}

\section{حساب لامبدا نوع‌دار ساده}

حساب لامبدا نوع‌دار ساده\footnote{\lr{Simply Typed Lambda Calculus} یا به اختصار \lr{STLC}}
یکی از مهم‌ترین و کلاسیک‌ترین سیستم‌های نوع است.
این سیستم پایه بسیاری از زبان‌های تابعی، سیستم‌های نوع پیشرفته و بخش مهمی از نظریه انواع به شمار می‌رود.
همچنین ساده‌ترین نمونه از «سیستم‌های نوع محض» است و مقدمه‌ای طبیعی برای ورود به مباحث بخش بعدی به حساب می‌آید.

در این بخش نحو، قضاوت‌ها و قواعد نوع STLC را معرفی می‌کنیم.

\subsection{نحو}

\begin{defn}[انواع]
مجموعه انواع به صورت بازگشتی زیر تعریف می‌شود
\begin{equation*}
T ::= B | T \to T
\end{equation*}
که در آن،
$B=\{\alpha,\beta,\gamma,...\}$
مجموعه توابع پایه و
$T \to T$
سازنده نوع تابع است.
\end{defn}

\begin{defn}[عبارات]
مجموعه عبارات به صورت بازگشتی زیر تعریف می‌شود
\begin{equation*}
M ::= V | \lambda V^T.M | MM
\end{equation*}
که در آن؛
$V=\{x,y,z,...\}$
مجموعه متغیرها،
$T$
مجموعه انواع،
$\lambda V^T.M$
انتزاع و
$MM$
کاربرد است.
\end{defn}

\subsection{متن و قضاوت نوع}

\begin{defn}[متن]
یک متن مانند
$\Gamma$،
دنباله‌ای متناهی از مفروضات به شکل زیر است
\begin{equation*}
x_1:A_1,...,x_n:A_n    
\end{equation*}
که در آن
$x_i:A_i$
به این معناست که
"$x_i$
از نوع
$A_i$
است".
\end{defn}

\begin{defn}[قضاوت نوع]
یک قضاوت نوع، گزاره‌ای به شکل زیر است
\begin{equation*}
\Gamma \vdash M:A
\end{equation*}
که در آن
$\Gamma$
یک متن،
$M$
یک عبارت و
$A$
یک نوع است و معنای آن این است که
"در متن
$\Gamma$،
عبارت
$M$
از نوع
$A$
است".
\end{defn}

\begin{defn}[قواعد قضاوت نوع]
قواعد قضاوت نوع در
$STLC$
شامل سه قاعده زیر است:

\begin{itemize}
\item قاعده متغیر
\begin{prooftree}
\AxiomC{}
\UnaryInfC{$\Gamma,x:A \vdash x:A$}
\end{prooftree}
\item قاعده انتزاع
\begin{prooftree}
\AxiomC{$\Gamma,x:A \vdash M:B$}
\UnaryInfC{$\Gamma \vdash \lambda x^A.M:A \to B$}
\end{prooftree}
\item قاعده کاربرد
\begin{prooftree}
\AxiomC{$\Gamma \vdash N:A$}
\AxiomC{$\Gamma \vdash M:A \to B$}
\BinaryInfC{$\Gamma \vdash MB:B$}
\end{prooftree}
\end{itemize}
\end{defn}
