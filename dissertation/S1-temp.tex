\section{مقدمه‌}

در این فصل، مفاهیم و تعاریف پایه‌ای مورد نیاز برای درک مباحث اصلی پایان‌نامه معرفی می‌شوند.
این مفاهیم شامل حساب لامبدا بدون نوع، سیستم‌های کاهش انتزاعی، حساب لامبدا نوع‌دار ساده و سیستم‌های نوع محض هستند که به ترتیب از ساده به پیچیده ارائه می‌گردند.
هدف از این فصل، ایجاد یک زبان مشترک و نمادگذاری واحد برای خواننده است تا زمینه لازم برای مطالعه فصل‌های بعدی، که به معرفی سیستم‌های نوع محض لایه‌ای، ترجمهٔ حذف وابستگی و صوری‌سازی آن‌ها در اثبات‌یار
\lr{Coq}
اختصاص دارد، فراهم آید.

فصل با بررسی حساب لامبدا بدون نوع آغاز می‌شود که پایه تاریخی و مفهومی تمام سیستم‌های بعدی را تشکیل می‌دهد.
سپس چارچوب کلی سیستم‌های کاهش انتزاعی معرفی می‌گردد که ابزار ریاضی برای مطالعه دقیق خواص نرمال‌سازی و همگرایی را فراهم می‌کند. این چارچوب به دلیل عمومیت خود، امکان بیان یکپارچه نتایج مربوط به نرمال‌سازی ضعیف، قوی و همگرایی را فراهم می‌کند.
در ادامه، حساب لامبدا نوع‌دار ساده به عنوان نخستین سیستم نوع‌دار وابسته بررسی می‌شود. این سیستم نه تنها مقدمه‌ای طبیعی برای ورود به سیستم‌های پیچیده‌تر است، بلکه نمونه‌ای کلاسیک از چگونگی جلوگیری انواع از حلقه‌های محاسباتی را نشان می‌دهد.
در نهایت، به سیستم‌های نوع محض می‌پردازیم که چارچوب کلی و یکپارچه‌ای را برای بیان طیف وسیعی از سیستم‌های نوع، از حساب لامبدا نوع‌دار ساده تا حساب ساختمان‌ها، ارائه می‌کنند.

اثبات‌های تفصیلی بسیاری از قضایا و لم‌های این فصل به منابع استاندارد مانند
\cite{barendregt} و \cite{geuvers1993}
ارجاع داده شده و تنها صورت تعاریف و قضایای کلیدی همراه با توضیحات ضروری ارائه می‌گردد.
این رویکرد از تکرار مطالب جلوگیری می‌کند و خواننده را به مطالعه منابع اصلی ترغیب می‌نماید.

با پایان این فصل، خواننده با ابزارهای نظری لازم برای درک سیستم‌های نوع محض لایه‌ای و ترجمه حذف وابستگی آشنا خواهد شد و آماده ورود به مباحث تخصصی فصل‌های بعدی می‌شود.

\section{حساب لامبدا بدون نوع}

حساب لامبدا بدون نوع که نخستین بار توسط
\lr{Alonzo Church}
در
\cite{church1932}
معرفی شد، پایه تاریخی و مفهومی تمام سیستم‌های محاسباتی تابعی و نظریه انواع مدرن به شمار می‌رود.
این دستگاه، محاسبه را به عنوان کاربرد توابع بر آرگومان‌ها مدل می‌کند و قدرت بیان معادلی با ماشین تورینگ دارد.
این حساب، با وجود سادگی نحوی، مسائل عمیقی مانند حلقه‌های نامتناهی و تصمیم‌ناپذیری را نشان می‌دهد؛ که این موارد در سیستم‌های نوع‌دار با استفاده از انواع کنترل می‌شوند.

\subsection{نحو}

\begin{defn}[نحو عبارات در حساب لامبدا بدون نوع]    
مجموعه عبارات حساب لامبدا بدون نوع، که معمولا با
$\Lambda$
نشان داده می‌شود، به صورت بازگشتی زیر تعریف می‌گردد

\begin{equation*}
M ::= V \mid \lambda V.M \mid MM
\end{equation*}

که در آن

\begin{itemize}
\item $V=\{x,y,z,\dots\}$ مجموعه متغیرهاست.
\item $\lambda V.M$ یک انتزاع و بیانگر تعریف یک تابع است.
\item $MM$ یک کاربرد و بیانگر صدا زدن یک تابع است.
\end{itemize}
\end{defn}

\subsection{متغیرهای آزاد و بسته}

برای درک دقیق مفاهیم جانشینی و کاهش، معرفی متغیرهای آزاد و بسته ضروری است.

\begin{defn}[متغیرهای آزاد یک عبارت]
برای یک عبارت
$M$،
مجموعه متغیرهای آزاد آن که با
$FV(M)$
نشان داده می‌شود، به صورت بازگشتی زیر تعریف می‌گردد
\begin{align*}
FV(x)&=\{x\}\\
FV(\lambda x.M)&=FV(M)-\{x\}\\
FV(MN)&=FV(M) \cup FV(N)
\end{align*}
\end{defn}

\begin{prop}
گزاره‌های زیر با یکدیگر معادل هستند:
\begin{itemize}
\item $x \in FV(M)$.
\item متغیر $x$ در عبارت $M$ آزاد است.
\item متغیر $x$ در عبارت $M$ دارای رخداد آزاد است.
\end{itemize}
\end{prop}

\begin{prop}
گزاره‌های زیر با یکدیگر معادل هستند:
\begin{itemize}
\item $x \notin FV(M)$.
\item متغیر $x$ در عبارت $M$ بسته است.
\item متغیر $x$ در عبارت $M$ دارای رخداد بسته است.
\end{itemize}
\end{prop}

\begin{prop}
گزاره‌های زیر با یکدیگر معادل هستند:
\begin{itemize}
\item $FV(M) = \emptyset$.
\item عبارت $M$ بسته است.
\item عبارت $M$ یک ترکیب‌گر است.
\end{itemize}
\end{prop}

\subsection{جانشینی}

جانشینی با نماد
$M[N/x]$
نشان داده می‌شود و معنای آن جایگزینی همه رخدادهای آزاد
$x$
در
$M$
با عبارت
$N$
است.

\begin{rem}
در برخی منابع مانند
\cite{barendregt}،
جانشینی با نماد
\begin{equation*}
M[x:=N]
\end{equation*}
نشان داده می‌شود. ما در این پایان‌نامه از نماد متداول دیگری استفاده می‌کنیم که
\lr{Mull}
نیز در تز دکتری خود استفاده کرده است، یعنی
\begin{equation*}
M[N/x]   
\end{equation*}
\end{rem}

\begin{defn}
جانشینی به صورت بازگشتی زیر تعریف می‌شود
\begin{align*}
x[N/x] &= N\\
y[N/x] &= y & (x \neq y) \\
PQ[N/x]&=(P[N/x])(Q[N/x])\\
(\lambda x.M)[N/x] &= \lambda x.M\\
(\lambda y.M)[N/x] &= \lambda y.(M[N/x]) & (x \neq y)
\end{align*}
\end{defn}

\subsection{تبدیل $\alpha$}

تبدیل
$\alpha$
به ما اجازه می‌دهد نام متغیرهای بسته را، بدون تغییر معنای عبارت، تغییر دهیم.

\begin{defn}
اگر
$M$
یک عبارت باشد و
$x \notin FV(M)$،
آنگاه تبدیل
$\alpha$
به صورت زیر بیان می‌شود
\begin{equation*}
\lambda x.M \equiv_{\alpha} \lambda y.M[y/x]
\end{equation*}
\end{defn}

\subsection{کاهش $\beta$}

\begin{defn}
کاهش
$\beta$
قاعده اصلی محاسبه در حساب لامبدا است و به صورت زیر تعریف می‌شود
\begin{equation*}
(\lambda x.M)N \rightarrow_{\beta} M[N/x]
\end{equation*}
که بستار بازتاب-تعدی آن با نماد
$\twoheadrightarrow_{\beta}$
نشان داده می‌شود.
\end{defn}

\subsubsection{مثال}
\begin{enumerate}
\item
\begin{align*}
\lambda x.(\lambda xy.y)x &\equiv_{\alpha} \lambda x.(\lambda ry.y)x\\
&\rightarrow_{\beta} \lambda x.(\lambda y.y)[x/r]\\
&= \lambda x.(\lambda y.y)\\
\end{align*}
\item \label{infinite_reduction} اگر $M \equiv \lambda x.xx$، آنگاه
\begin{align*}
MM &\equiv (\lambda x.xx)M\\
&\rightarrow_{\beta} xx[M/x]\\
&= MM\\
&\equiv (\lambda x.xx)M\\
&\rightarrow_{\beta} xx[M/x]\\
&= \dots
\end{align*}
پس داریم
\begin{equation*}
MM \rightarrow_{\beta} MM \rightarrow_{\beta} MM \rightarrow_{\beta} \dots
\end{equation*}
\end{enumerate}

همانطور که در مثال ۲ مشاهده می‌کنید، دنباله کاهش عبارت $MM$ نامتناهی است. این مثال ما را به مفاهیم بعدی سوق می‌دهد.

\subsection{مسیرهای کاهش و فرم نرمال}

\begin{defn}
یک مسیر کاهش، دنباله‌ای به شکل زیر است که می‌تواند متناهی یا نامتناهی باشد.
\begin{equation*}
M_0 \rightarrow_{\beta} M_1 \rightarrow_{\beta} \dots \rightarrow_{\beta} M_n
\end{equation*}
\end{defn}

\begin{defn}
عبارت
$M$
در فرم نرمال (یا یک فرم نرمال) است، اگر هیچ گامی از کاهش
$\beta$
بر آن قابل اعمال نباشد.
یعنی
\begin{equation*}
M \in \mathrm{NF} \iff \nexists N. M \rightarrow_{\beta} N
\end{equation*}
\end{defn}

\subsubsection{مثال}
عبارات زیر در فرم نرمال هستند:
\begin{align*}
&x\\
&x(\lambda a.a)\\
&\lambda x.x\\
&\lambda y.x
\end{align*}

عبارات زیر در فرم نرمال نیستند:
\begin{align*}
&(\lambda x.x)x\\
&(\lambda y.xy)(\lambda z.r)
\end{align*}

زیرا به ترتیب داریم:
\begin{align*}
&(\lambda x.x)x \equiv_{\alpha} (\lambda y.y)x \rightarrow_{\beta} y[x/y] = x\\
&(\lambda y.xy)(\lambda z.r) \rightarrow_{\beta} xy[(\lambda z.r)/y] = x(\lambda z.r)
\end{align*}

\subsection{نرمال‌سازی ضعیف و قوی}

\begin{defn}[نرمال‌سازی ضعیف]
عبارت
$M$
دارای خاصیت نرمال‌سازی ضعیف است، اگر حداقل یک مسیر کاهش به یک فرم نرمال داشته باشد.
یعنی
\begin{equation*}
\exists N\in \mathrm{NF}.\;M \twoheadrightarrow_{\beta} N
\end{equation*}
\end{defn}

\begin{defn}[نرمال‌سازی قوی]
عبارت
$M$
دارای خاصیت نرمال‌سازی قوی است، اگر هیچ مسیر کاهش نامتناهی‌ای از آن آغاز نشود.
\end{defn}

\begin{thm}
اگر یک عبارت دارای خاصیت نرمال‌سازی قوی باشد، آنگاه دارای خاصیت نرمال‌سازی ضعیف نیز است.
\end{thm}

عکس قضیه فوق همان حدس
\lr{BGK}
است که برای سیستم‌های وابسته، هنوز اثبات نشده است.

\begin{conj}[\lr{BGK}]\label{BGK}
اگر یک عبارت دارای خاصیت نرمال‌سازی ضعیف باشد، آنگاه دارای خاصیت نرمال‌سازی قوی نیز است.
\end{conj}

\begin{thm}
در حساب لامبدا بدون نوع، نرمال‌سازی ضعیف و قوی معادل‌ هستند؛ به بیان دیگر، حدس
\ref{BGK}
در حساب لامبدا بدون نوع، برقرار است.
\end{thm}

\subsubsection{مثال}
همانطور که در
\ref{infinite_reduction}
دیدیم، عبارت زیر دارای خاصیت نرمال‌سازی قوی نیست
\begin{equation*}
(\lambda x.xx)(\lambda x.xx)
\end{equation*}

\section{سیستم‌های کاهش انتزاعی}

سیستم‌های کاهش انتزاعی\footnote{\lr{Abstract Reduction Systems} یا به اختصار \lr{ARS}}
چارچوبی ریاضی برای مطالعه روابط کاهش در سیستم‌های محاسباتی هستند.
بسیاری از مفاهیم نظریه انواع
(از جمله نرمال‌سازی ضعیف و قوی، همگرایی و یکتایی فرم نرمال)
به‌طور طبیعی در این چارچوب بیان می‌شوند.
در این بخش تعریف رسمی و ویژگی‌های بنیادی این چارچوب را بیان می‌کنیم.

\subsection{تعریف سیستم کاهش انتزاعی}

\begin{defn}
یک سیستم کاهش انتزاعی، یک زوج
$(A,\rightarrow)$
است که در آن:
\begin{itemize}
\item 
$A$
مجموعه‌ اشیا است.
\item
$\rightarrow \; \subseteq A \times A$
یک رابطه دودویی روی
$A$
است که رابطه کاهش نام دارد.
\end{itemize}
\end{defn}

برای مثال در حساب لامبدا، مجموع عبارات
$\Lambda$
همراه با رابطه کاهش
$\beta$،
یک
ARS
را تشکیل می‌دهند.

\subsection{دنباله و بستار کاهش}

\begin{defn}
دنباله کاهش، زنجیره‌ای از کاهش‌ها به شکل
$a_0 \rightarrow a_1 \rightarrow \dots \rightarrow a_n$
است، که می‌تواند متناهی یا نامتناهی باشد.
\end{defn}

\begin{defn}
بستار بازتابی-تعدی رابطه
$\rightarrow$
را با
$b \rightarrow^* a$
نشان می‌دهیم که یعنی
$b = a$
یا
دنباله کاهشی از
$b$
به
$a$
وجود دارد.
\end{defn}

\subsection{فرم نرمال و نرمال‌سازی}

\begin{defn}
شی
$a$
یک فرم نرمال است اگر
$\nexists b.\;a \rightarrow b$.
\end{defn}

\begin{defn}
شی
$a$
دارای خاصیت نرمال‌سازی ضعیف است، اگر
$\exists b.\;(a \rightarrow^* b \land b \in \mathrm{NF})$.
\end{defn}

\begin{defn}
شی
$a$
دارای خاصیت نرمال‌سازی قوی است، اگر هیچ دنباله کاهش نامتناهی از آن آغاز نشود.
\end{defn}

\begin{rem}
بدیهی است که
\lr{$\text{Strong Normalization} \implies \text{Weak Normalization}$}،
اما عکس آن همیشه برقرار نیست.
\end{rem}

\subsection{همگرایی}

\begin{defn}[همگرایی]
سیستم کاهش انتزاعی
$(A,\rightarrow)$
همگراست هرگاه
\begin{equation*}
\forall a,b_1,b_2.\;(a \rightarrow^* b_1 \land a \rightarrow^* b_2) \implies (\exists c.\; (b_1 \rightarrow^* c \land b_2 \rightarrow^* c))
\end{equation*}
به بیان ساده؛ هر دو مسیر کاهش که از یک شی منشعب می‌شوند، دوباره به شی مشترکی می‌رسند.
\end{defn}

نتایجی مانند یکتایی فرم نرمال از قضیه پیش رو ثابت می‌شوند.

\begin{thm}[قضیه چرچ-راسر]
سیستم کاهش انتزاعی
$(A,\rightarrow)$
همگراست اگر و تنها اگر دارای خاصیت چرچ-راسر باشد؛ یعنی
\begin{equation*}
\forall b_1,b_2.\;(b_1 = b_2) \implies (\exists c.\; (b_1 \rightarrow^* c \land b_2 \rightarrow^* c))
\end{equation*}
\end{thm}

اثبات همگرایی معمولاً دشوار است، اما یک ویژگی محلی ساده‌تر وجود دارد.

\begin{defn}[همگرایی محلی]
سیستم کاهش انتزاعی
$(A,\rightarrow)$
همگرای محلی است اگر
\begin{equation*}
\forall a,b_1,b_2.\;(a \rightarrow b_1 \land a \rightarrow b_2) \implies (\exists c.\; (b_1 \rightarrow^* c \land b_2 \rightarrow^* c))
\end{equation*}
\end{defn}

\begin{lmm}[لم نیومن]
اگر یک سیستم کاهش انتزاعی،همگرای محلی و دارای خاصیت نرمال‌سازی قوی باشد، آنگاه همگراست.
\end{lmm}

\section{حساب لامبدا نوع‌دار ساده}

حساب لامبدا نوع‌دار ساده\footnote{\lr{Simply Typed Lambda Calculus} یا به اختصار \lr{STLC}}
یکی از مهم‌ترین و کلاسیک‌ترین سیستم‌های نوع است.
این سیستم پایه بسیاری از زبان‌های تابعی، سیستم‌های نوع پیشرفته و بخش مهمی از نظریه انواع به شمار می‌رود.
همچنین ساده‌ترین نمونه از «سیستم‌های نوع محض» است و مقدمه‌ای طبیعی برای ورود به مباحث بخش بعدی به حساب می‌آید.

در این بخش نحو، قضاوت‌ها و قواعد نوع STLC را معرفی می‌کنیم.

\subsection{نحو}

\begin{defn}[انواع]
مجموعه انواع به صورت بازگشتی زیر تعریف می‌شود
\begin{equation*}
T ::= B \mid T \to T
\end{equation*}
که در آن،
$B=\{\alpha,\beta,\gamma,\dots\}$
مجموعه توابع پایه و
$T \to T$
سازنده نوع تابع است.
\end{defn}

\begin{defn}[عبارات]
مجموعه عبارات به صورت بازگشتی زیر تعریف می‌شود
\begin{equation*}
M ::= V \mid \lambda V^T.M \mid MM
\end{equation*}
که در آن؛
$V=\{x,y,z,\dots\}$
مجموعه متغیرها،
$T$
مجموعه انواع،
$\lambda V^T.M$
انتزاع و
$MM$
کاربرد است.
\end{defn}

\subsection{زمینه و قضاوت نوع}

\begin{defn}[زمینه]
یک زمینه مانند
$\Gamma$،
دنباله‌ای متناهی از مفروضات به شکل زیر است
\begin{equation*}
x_1:A_1,\dots,x_n:A_n    
\end{equation*}
که در آن
$x_i:A_i$
به این معناست که
"$x_i$
از نوع
$A_i$
است".

همچنین
$\Gamma,x:A$
به معنای افزودن فرض
$x:A$
به
$\Gamma$
است.
\end{defn}

\begin{defn}[قضاوت نوع]
یک قضاوت نوع، گزاره‌ای به شکل زیر است
\begin{equation*}
\Gamma \vdash M:A
\end{equation*}
که در آن
$\Gamma$
یک زمینه،
$M$
یک عبارت و
$A$
یک نوع است و معنای آن این است که
"در زمینه
$\Gamma$،
عبارت
$M$
از نوع
$A$
است".
\end{defn}

\begin{defn}[قواعد قضاوت نوع]
قواعد قضاوت نوع در
$STLC$
شامل سه قاعده زیر است:

\begin{itemize}
\item قاعده متغیر
\begin{prooftree}
\AxiomC{}
\UnaryInfC{$\Gamma,x:A \vdash x:A$}
\end{prooftree}
\item قاعده انتزاع
\begin{prooftree}
\AxiomC{$\Gamma,x:A \vdash M:B$}
\UnaryInfC{$\Gamma \vdash \lambda x^A.M:A \to B$}
\end{prooftree}
\item قاعده کاربرد
\begin{prooftree}
\AxiomC{$\Gamma \vdash N:A$}
\AxiomC{$\Gamma \vdash M:A \to B$}
\BinaryInfC{$\Gamma \vdash MB:B$}
\end{prooftree}
\end{itemize}
\end{defn}

\section{سیستم‌های نوع محض}

سیستم‌های نوع محض\footnote{\lr{Pure Type Systems} یا به اختصار \lr{PTS}}
که نخستین بار توسط
\cite{berardi1988} و \cite{terlouw1989}
معرفی شدند و سپس توسط
\cite{barendregt1991}
تعمیم یافتند، یک چارچوب یکپارچه برای توصیف طیف وسیعی از سیستم‌های نوع مبتنی بر حساب لامبدا را فراهم می‌کنند.
بسیاری از سیستم‌های مهم از جمله حساب لامبدا نوع‌دار ساده، سیستم‌های چندریخت و حساب ساختمان‌ها را می‌توان تنها با انتخاب مناسب مجموعه‌ای از "رده‌ها"، "اصول" و "روابط" در قالب یک
\lr{PTS}
بیان کرد. این چارچوب به دلیل سادگی نحوی و هم‌چنین انعطاف بسیار بالا، نقش محوری در نظریه انواع دارد. از طرفی سیستمی که
\lr{Mull}
در \cite{mull2023}
معرفی می‌کند که اساس کار این پایان‌نامه است، نسخه‌ تغییر یافته همین سیستم است؛
لذا آشنایی با این سیستم برای ما ضروری است.

\subsection{مشخصه و نحو}

\begin{defn}[مشخصه سیستم نوع محض]
یک سیستم نوع محض با سه تایی
$(\mathcal{S},\mathcal{A},\mathcal{R})$
مشخص می‌شود که در آن
\begin{itemize}
\item $S=\{s_1,s_2,\dots\}$ مجموعه رده‌ها
\item $A \subseteq S \times S$ مجموعه اصول
\item $R \subseteq S \times S \times S$ مجموعه روابط
\end{itemize}
هستند.
\end{defn}

عبارات و انواع در
\lr{PTS}
از منظر نحو یکسان هستند؛یعنی در این چارچوب، "نوع" و "عبارت" یک ساختار واحد دارند و فقط جایگاه آن‌ها در قضاوت نوع است که نقش متفاوتی به آن‌ها می‌دهد.

\begin{defn}[نحو سیستم نوع محض]
نحو سیستم نوع محض به صورت بازگشتی زیر تعریف می‌شود
\begin{equation*}
T ::= \mathcal{S} \mid V \mid \lambda V:T.T \mid \Pi V:T.T \mid TT
\end{equation*}
که در آن
\begin{itemize}
\item $\mathcal{S}=\{s_1,s_2,\dots\}$ مجموعه رده‌ها
\item $V=\{x,y,z,\dots\}$ مجموعه متغیرها
\item $\lambda V:T.T$ انتزاع
\item $\Pi V:T.T$ نوع تابع
\item $TT$ کاربرد
\end{itemize}
هستند.
\end{defn}

در این پایان‌نامه، مطابق با
\cite{mull2023}،
متغیرها را با رده خودشان نشانه‌گذاری می‌شوند.

\begin{defn}[نشانه گذاری متغیرها]
برای هر
$s \in \mathcal{S}$، 
$V_s$
را مجموعه شمارشی متغیرهای مربوط به رده
$s$
در نظر می‌گیریم.

$^sv_i$
به معنای
$i$
مین متغیر در
$V_s$
است و
$V = \cup_{s \in \mathcal{S}}V_s$.
\end{defn}

\subsection{قضاوت نوع}

\begin{defn}[زمینه]
یک زمینه مانند
$\Gamma$،
دنباله‌ای متناهی از مفروضات به شکل زیر است
\begin{equation*}
x_1:A_1,\dots,x_n:A_n    
\end{equation*}
که در آن
$x_i:A_i$
به این معناست که
"$x_i$
از نوع
$A_i$
است".
همچنین
$\Gamma,x:A$
به معنای افزودن فرض
$x:A$
به
$\Gamma$
است.
\end{defn}

\begin{defn}[قواعد قضاوت نوع در سیستم نوع محض]
در یک سیستم نوع محض با مشخصه
$(\mathcal{S},\mathcal{A},\mathcal{R})$
قواعد زیر موجود است:

\begin{itemize}
\item قاصده اصل
\begin{prooftree}
\AxiomC{}
\LeftLabel{$\quad (s,t) \in \mathcal{A}$}
\UnaryInfC{$\vdash s:t$}
\end{prooftree}
\item قاعده شروع
\begin{prooftree}
\AxiomC{$\Gamma \vdash A:s$}
\LeftLabel{$\quad x \notin \Gamma$}
\UnaryInfC{$\Gamma,x:A \vdash x:A$}
\end{prooftree}
\item قاعده تضعیف
\begin{prooftree}
\AxiomC{$\Gamma \vdash B:s$}
\AxiomC{$\Gamma \vdash M:A$}
\LeftLabel{$\quad x \notin \Gamma$}
\BinaryInfC{$\Gamma,x:B \vdash M:A$}
\end{prooftree}
\item قاعده تولید
\begin{prooftree}
\AxiomC{$\Gamma,x:A \vdash B:s_2$}
\AxiomC{$\Gamma \vdash A:s_1$}
\LeftLabel{$\quad (s_1,s_2,s_3) \in \mathcal{R}$}
\BinaryInfC{$\Gamma \vdash (\Pi x:A.B):s_3$}
\end{prooftree}
\item قاعده انتزاع
\begin{prooftree}
\AxiomC{$\Gamma \vdash \Pi x:A.B:s$}
\AxiomC{$\Gamma,x:A \vdash M:B$}
\BinaryInfC{$\Gamma \vdash (\lambda x:A.M):(\Pi x:A.B)$}
\end{prooftree}
\item قاعده کاربرد
\begin{prooftree}
\AxiomC{$\Gamma \vdash N:A$}
\AxiomC{$\Gamma \vdash M:(\Pi x:A.B)$}
\BinaryInfC{$\Gamma \vdash MN:B[N/x]$}
\end{prooftree}
\item قاعده تبدیل
\begin{prooftree}
\AxiomC{$\Gamma \vdash B:s$}
\AxiomC{$A \equiv_\beta B$}
\AxiomC{$\Gamma \vdash M:A$}
\TrinaryInfC{$\Gamma \vdash M:B$}
\end{prooftree}
\end{itemize}
\end{defn}

\subsection{نمونه‌ها}

حال با معرفی کامل
\lr{PTS}،
بسیاری از سیستم‌های معروف تنها با انتخاب مناسب
$(\mathcal{S},\mathcal{A},\mathcal{R})$
قابل بیان‌اند.
\\ \\
\textbf{حساب لامبدا نوع‌دار ساده}
\begin{align*}
\mathcal{S} &= \{*\}\\
\mathcal{A} &= \{(*,*)\}\\
\mathcal{R} &= \{(*,*,*)\}
\end{align*}
\\
\textbf{سیستم $\lambda2$}
\begin{align*}
\mathcal{S} &= \{*,\Box\}\\
\mathcal{A} &= \{(*,\Box)\}\\
\mathcal{R} &= \{(*,*,*),(*,*,\Box)\}
\end{align*}
\\
\textbf{سیستم $\lambda_\omega$}
\begin{align*}
\mathcal{S} &= \{*,\Box\}\\
\mathcal{A} &= \{(*,\Box)\}\\
\mathcal{R} &= \{(*,*,*),(*,*,\Box),(\Box,\Box,\Box)\}
\end{align*}

\subsection{مکعب لامبدا}

مکعب لامبدا یک چارچوب مفهومی و هندسی برای دسته‌بندی خانواده‌ای از سیستم‌های نوع محض است. این مکعب نشان می‌دهد که چگونه با افزودن یا حذف برخی قابلیت‌های نوعی، می‌توان به سیستم‌های مختلفی از حساب لامبدا دست یافت.

\begin{figure}[h!]
\centering
\includegraphics[scale=0.2]{lambda_cube.png}
\caption{مکعب لامبدا}
\label{fig.lambda_cube}
\end{figure}

مکعب لامبدا دارای سه بُعد مستقل است که هر بُعد متناظر با افزودن یک قابلیت به سیستم پایه است:

\begin{enumerate}
\item \textbf{چندریختی}\\
این بُعد نشان‌دهنده امکان کمی‌سازی روی انواع است؛ به بیان دیگر، اجازه می‌دهد توابعی تعریف شوند که برای همه انواع معتبر باشند.
وجود این قابلیت منجر به سیستم‌هایی مانند
$\lambda2$
می‌شود.
\item \textbf{اپراتورهای نوع}\\
این بُعد امکان تعریف توابعی را فراهم می‌کند که روی انواع عمل می‌کنند، نه روی عبارات. در این حالت، انواع می‌توانند به‌عنوان ورودی توابع به‌کار روند و این امر منجر به سیستم‌هایی مانند
$\lambda_\omega$
می‌شود.
\item \textbf{انواع وابسته}\\
این بُعد اجازه می‌دهد نوع یک عبارت به مقدار یک عبارت دیگر وابسته باشد. به این ترتیب می‌توان انواعی تعریف کرد که اطلاعاتی درباره مقدار عبارات در خود جای می‌دهند؛ این قابلیت، پایه سیستم‌هایی مانند حساب ساختمان‌هاست.
\end{enumerate}

ترکیب یا عدم ترکیب هر یک از این سه قابلیت، هشت رأس مکعب را تشکیل می‌دهد که هر رأس متناظر با یک سیستم نوع مشهور است. برای مثال، حساب لامبدای نوع‌دار ساده در رأسی قرار دارد که هیچ‌یک از این قابلیت‌ها را ندارد، در حالی که حساب ساختمان‌ها در رأسی قرار می‌گیرد که هر سه قابلیت را به‌طور هم‌زمان داراست.



به این ترتیب، مکعب لامبدا ابزاری شهودی برای مقایسه قدرت بیانی و ساختار سیستم‌های نوع محض فراهم می‌کند و نقش مهمی در درک ارتباط میان این سیستم‌ها ایفا می‌کند.

\subsection{اهمیت}

در ادامه این پایان‌نامه، سیستم‌های نوع محض نقطهٔ آغاز برای تعریف «سیستم‌های نوع محض لایه‌ای\footnote{\lr{Tiered Pure Type Systems} یا \lr{TPTS}}» 
سپس معرفی ترجمهٔ حذف وابستگی \lr{Mull} هستند.
در واقع:
\begin{itemize}
\item \lr{TPTS} گسترشی ساختاریافته از \lr{PTS} است.
\item قواعد نوع \lr{TPTS} نسخه تقویت‌شده همین قواعد‌اند.
\item اثبات‌های نرمال‌سازی وابسته به ساختار \lr{PTS} بیان می‌شوند.
\end{itemize}
در نتیجه، این بخش چارچوب مفهومی لازم برای تعریف سیستم‌های لایه‌ای و صوری‌سازی آن‌ها در \lr{Coq} را فراهم می‌کند.
