\begin{frame}{مسئله اصلی}

\textbf{مسئله نرمال‌سازی (حدس \lr{BGK}\footnote{\lr{Barendregt–Geuvers–Klop}})}\\
در هر سیستم نوع محض\footnote{\lr{PTS}}؛
نرمال‌سازی ضعیف
$\overset{?}{\impliedby}$
نرمال‌سازی قوی
\hfil\\
\hfil\\
{\small
  \begin{itemize}
      \item \emph{فرم نرمال}: عبارتی که قابل کاهش نباشد.
      \item \emph{نرمال‌سازی ضعیف}: وجود مسیر کاهش به فرم نرمال برای هر عبارت نوع‌پذیر.
      \item \emph{نرمال‌سازی قوی}: عدم وجود دنباله کاهش نامتناهی برای هر عبارت نوع‌پذیر.
  \end{itemize}
}

\end{frame}

\begin{frame}{چرا این مسئله دشوار است؟}
در سیستم‌های وابسته؛
\begin{itemize}
  \item انواع به عبارات وابسته‌اند.
  \item کاهش روی نوع اثر می‌گذارد.
  \item جانشینی بسیار ظریف و حساس می‌شود.
\end{itemize}
\end{frame}

\begin{frame}{ایده \lr{Nathan Mull}}

\textbf{سیستم‌های نوع محض لایه‌ای\footnote{\lr{TPTS}}}\\
\hfil{}\\
ایده‌های کلیدی:
{\small
  \begin{itemize}
    \item اختصاص درجه به انواع و عبارات
    \item محدودسازی وابستگی بین درجات
    \item تعریف ترجمه‌ای برای حذف وابستگی
  \end{itemize}
}

\textbf{نتیجه}\\
نرمال‌سازی قوی در سیستم مقصد
$\impliedby$
نرمال‌سازی قوی در سیستم مبدا

\end{frame}
